\begin{lemma}[Closure Monotonicity]
\label{lem:topological-closure-monotone}
Let \(X\) be a topological space, and let \(A,B\subseteq X\).  If
\(A\subseteq B\), then
\[
\mathrm{Cl}_{X}(A)\subseteq\mathrm{Cl}_{X}(B).
\]
